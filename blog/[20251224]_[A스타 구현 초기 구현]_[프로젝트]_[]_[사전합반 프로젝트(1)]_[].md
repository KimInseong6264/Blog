```
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Tilemaps;

public class Astar
{
    private Tilemap _wallTilemap;
    private NodeManager _nodeManager;

    private List<Node> _openList;
    private Dictionary<Vector2Int, Node> _openDict;
    private HashSet<Vector2Int> _closedSet;
    private int _count;

    public Astar(Tilemap wallTilemap)
    {
        _wallTilemap = wallTilemap;
        _nodeManager = new NodeManager(wallTilemap);

        _openList = new List<Node>();
        _openDict = new Dictionary<Vector2Int, Node>();
        _closedSet = new HashSet<Vector2Int>();
    }


    // 경로 탐색 알고리즘
    public List<Vector2> Pathfinder(Vector3 start, Vector3 target)
    {
        Init();

        Vector2Int startCellPos = (Vector2Int)_wallTilemap.WorldToCell(start);
        Vector2Int targetCellPos = (Vector2Int)_wallTilemap.WorldToCell(target);
        
        if (startCellPos == targetCellPos)
            return null;

        Node current = _nodeManager.GetNode(startCellPos, 0, Heuristic(startCellPos, targetCellPos));

        _openList.Add(current);
        _openDict.Add(startCellPos, current);


        while (_count < 300 && _openList.Count > 0)
        {
            _count++;

            current = GetLowestFNode(_openList);

            // 타겟에 도달시, BuildPath로 경로 출력
            if (current.Pos == targetCellPos)
                return BuildPath(current);

            _openList.Remove(current);
            _openDict.Remove(current.Pos);
            _closedSet.Add(current.Pos);

            // 갈 수 있는 이웃 셀좌표를 GetNeighbors로 생성 후 순회하며 이웃 노드 추가
            foreach (var neighborPos in _nodeManager.GetNeighbors(current.Pos))
            {
                // 방문 기록이 있다면 다음 이웃 노드 확인
                if (_closedSet.Contains(neighborPos))
                    continue;

                int sumG = current.G + _nodeManager.GetMoveCost(current.Pos);

                // _openDict에 이웃 노드 정보가 없으면 이웃 노드 생성 및 추가
                if (!_openDict.TryGetValue(neighborPos, out Node neighborNode))
                {
                    neighborNode = _nodeManager.GetNode
                    (
                    neighborPos,
                    sumG,
                    Heuristic(neighborPos, targetCellPos),
                    current
                    );

                    _openList.Add(neighborNode);
                    _openDict.Add(neighborPos, neighborNode);

                }
                else if (sumG < neighborNode.G)
                {
                    neighborNode.G = sumG;
                    neighborNode.Parent = current;
                }
            }
        }

        Debug.LogWarning("경로 탐색 실패");
        return null;
    }


    // 최종 최적경로 출력 메서드
    List<Vector2> BuildPath(Node targetNode)
    {
        List<Vector2> path = new List<Vector2>();
        Node current = targetNode;

        while (current != null)
        {
            if(current.Parent != null)

            // 타일 중앙 좌표로 변환
            path.Add(GetCellPos(current.Pos));
            current = current.Parent;
        }

        path.Reverse();

        return path;
    }

    // 오픈리스트에서 F 비용이 가장 적은 노드를 반환
    private Node GetLowestFNode(List<Node> openList)
    {
        Node best = openList[0];

        foreach (var node in openList)
        {
            if (node.F < best.F || (node.F == best.F && node.H < best.H))
                best = node;
        }

        return best;
    }

    // 휴리스틱 예측치 계산
    private int Heuristic(Vector2Int current, Vector2Int target) =>
        (Mathf.Abs(target.x - current.x) + Mathf.Abs(target.y - current.y)) * 10;


    // 타일의 중앙으로 벡터값을 변경하는 오버로딩
    public Vector2 GetCellPos(Vector3 pos) => CellPos(pos);
    public Vector2 GetCellPos(Vector2 pos) => CellPos(pos);
    public Vector2 GetCellPos(Vector2Int pos) => CellPos(pos);

    private Vector2 CellPos(Vector2 pos) => 
        _wallTilemap.GetCellCenterWorld(Vector3Int.FloorToInt(pos));

    private void Init()
    {
        _count = 0;
        _openList.Clear();
        _openDict.Clear();
        _closedSet.Clear();
        _nodeManager.Init();
    }

}
```

public class NodeManager
{
    private Tilemap _wallTilemap;
    private List<Vector2Int> _neighborDic;

    // 좌표에 따른 노드들을 기억해 놓는 Dictionary
    private Dictionary<Vector2Int, Node> _nodeMap;

    public NodeManager(Tilemap wall)
    {
        _wallTilemap = wall;
        _neighborDic = new List<Vector2Int>();
        _nodeMap = new Dictionary<Vector2Int, Node>();
    }

    // 이동이 가능한 cell인지를 검사
    public bool IsWalkable(Vector2Int cellPos) => !_wallTilemap.HasTile((Vector3Int)cellPos);

    // 해당 셀의 코스트(G값)를 반환
    public int GetMoveCost(Vector2Int cellPos) => 10;


    // 해당 셀의 이웃한 셀(상하좌우)을 출력
    public Vector2Int[] GetNeighbors(Vector2Int cellPos)
    {
        // 이웃 노드를 임시로 담는 _neighborDic이 항상 초기화가 되어야 함
        _neighborDic.Clear();

        Vector2Int[] directions =
        {
        Vector2Int.up,
        Vector2Int.down,
        Vector2Int.left,
        Vector2Int.right
        };

        // 이웃한 타일이 갈 수 있는 타일인지 검사
        foreach (var dir in directions)
        {
            Vector2Int next = cellPos + dir;

            if (!IsWalkable(next))
                continue;

            _neighborDic.Add(next);
        }
        return _neighborDic.ToArray();
    }

    // 셀좌표에 해당하는 노드 생성
    public Node GetNode(Vector2Int cellPos, int g, int h, Node parent = null)
    {
        if (!_nodeMap.TryGetValue(cellPos, out var node))
        {
            // 셀좌표에 해당하는 노드가 없다면 새로 생성
            node = new Node(cellPos, g, h, parent);
            _nodeMap[cellPos] = node;
        }
        else if (g < node.G)
        {
            node.G = g;
            node.H = h;
            node.Parent = parent;
        }
        return node;
    }

    public void Init()
    {
        _nodeMap.Clear();
        _neighborDic.Clear();
    }
}

public class Node
{
    // 경로 탐색 시에 탐색할(or탐색한) 셀 좌표 구간을 Node로 지정한다.
    // 현재 타일에서 계산을 통해 다음 노드를 찾아나간다.

    public Vector2Int Pos { get; set; }   // 타일맵 셀 좌표
    public int G { get; set; }            // 시작 → 현재의 거리적 비용
    public int H { get; set; }            // 현재 → 목표의 휴리스틱
    public int F => G + H;                // 총 비용
    public Node Parent { get; set; }      // 경로 추적용 부모 노드

    public Node(Vector2Int pos, int g, int h, Node parent = null)
    {
        Pos = pos;
        G = g;
        H = h;
        Parent = parent;
    }
}

## 타일맵과 A스타 알고리즘

#### 노드 정의
A스타 알고리즘의 핵심은 거리와 예측치를 기반으로 최적의 경로를 찾는 알고리즘이다. <br>
이를 위해 가장 먼저 설정해야 하는 것은 **노드 설정**이다.

2D 프로젝트에서 노드를 사용하기 위해서는 **타일맵의 노드화**가 필요했다. <br>
먼저는 A스타 알고리즘을 위해서는 타일맵 정보가 필요하고, 이를 위해 생성자에서 해당 타일맵의 정보를 받아온다. <br>

이후에 타일맵의 셀좌표(이 때, 월드좌표와 셀좌표를 구분할 수 있어야 한다.)를 기반으로 노드를 정의한다. <br>
(노드의 셀좌표, G(거리), H(휴리스틱 예측치), 경로 역탐색을 위한 부모 노드)


#### 노드 매니저
다음 단계로는 노드 매니저를 통해 A스타에서 사용할 노드들의 생성과 규칙을 정의한다.

##### 1) 해당 셀좌표에 해당하는 노드를 저장하는 Dictionary
```
private Dictionary<Vector2Int, Node> _nodeMap;
```
해당 셀좌표를 키값에 해당하는 노드를 보관한다. <br>
이 때 이미 셀좌표의 노드가 _nodeMap에 존재시에는 Dictionary에서 반환해준다.(셀좌표 중복 입력 방지, **초기화 반드시 필요**)

##### 2) 입력한 노드가 갈 수 있는 노드인지 판단
```
public bool IsWalkable(Vector2Int cellPos) => !_wallTilemap.HasTile((Vector3Int)cellPos);
```
벽타일이 깔려있는 경우는 갈 수 없다는 false를 반환한다. <br>
(확실한 경로탐색을 위해 그라운드 타일인지도 확인하는 것이 좋음)

##### 3) 입력받은 노드를 기준으로 이웃 노드를 반환
```
public Vector2Int[] GetNeighbors(Vector2Int cellPos)
{
    // 이웃 노드를 임시로 담는 _neighborDic이 항상 초기화가 되어야 함
    _neighborDic.Clear();

    Vector2Int[] directions =
    {
    Vector2Int.up,
    Vector2Int.down,
    Vector2Int.left,
    Vector2Int.right
    };

    // 이웃한 타일이 갈 수 있는 타일인지 검사
    foreach (var dir in directions)
    {
        Vector2Int next = cellPos + dir;

        if (!IsWalkable(next))
            continue;

        _neighborDic.Add(next);
    }
    return _neighborDic.ToArray();
}
```
입력 노드의 상하좌우 노드를 반환한다. 이때, 갈 수 없는 노드 판단하여 반환한다. <br>
(반환하는 이웃 노드 _neighborDic도 **탐색시마다 반드시 초기화 필요**)

##### 4) 노드 매니저에서 벡터에 해당하는 노드 반환
```
public Node GetNode(Vector2Int cellPos, int g, int h, Node parent = null)
{
    if (!_nodeMap.TryGetValue(cellPos, out var node))
    {
        // 셀좌표에 해당하는 노드가 없다면 새로 생성
        node = new Node(cellPos, g, h, parent);
        _nodeMap[cellPos] = node;
    }
    else if (g < node.G)
    {
        node.G = g;
        node.H = h;
        node.Parent = parent;
    }
    return node;
}
```
_nodeMap에 입력 셀좌표가 들어있다면 _nodeMap에서 반환, 아닐 경우 새롭게 생성해서 추가 및 반환 <br>
(_nodeMap에 이미 있던 경우, 파라미터 g값이 더 적은 비용이라면 노드를 갱신한다.)


#### A스타 알고리즘 로직

##### 1) 현재 노드를 등록
```
Vector2Int startCellPos = (Vector2Int)_wallTilemap.WorldToCell(start);
Vector2Int targetCellPos = (Vector2Int)_wallTilemap.WorldToCell(target);

if (startCellPos == targetCellPos)
    return null;

Node current = _nodeManager.GetNode(startCellPos, 0, Heuristic(startCellPos, targetCellPos));

_openList.Add(current);
_openDict.Add(startCellPos, current);
```
A스타 내부에서는 **셀좌표로 탐색을 진행**한다. 현재 위치의 노드를 생성하고, 오픈 리스트와 오픈Dict에 추가한다. <br>
- 오픈 리스트: 갈 수 있는지 열어본 노드(현재 이동한 노드는 아님)
- 오픈 Dict: 해당 벡터에 해당하는 노드를 빠르게 열람하기 위한 Dictionary

##### 2) 오픈 리스트에서 가장 적은 F값의 노드로 이동
```
current = GetLowestFNode(_openList);

// 타겟에 도달시, BuildPath로 경로 출력
if (current.Pos == targetCellPos)
    return BuildPath(current);

_openList.Remove(current);
_openDict.Remove(current.Pos);
_closedSet.Add(current.Pos);
```
```
private Node GetLowestFNode(List<Node> openList)
{
    Node best = openList[0];

    foreach (var node in openList)
    {
        if (node.F < best.F || (node.F == best.F && node.H < best.H))
            best = node;
    }

    return best;
}
```
현재 노드가 가장 F값이 적은 노드로 이동후, 오픈리스트에서는 제외하고 _closedSet에 추가 <br>
(**HashSet**은 입력 파라미터의 값이 있는지만을 반환하는 빠른 열람에 유리)

##### 3) 이웃 노드를 확인해서 오픈 리스트에 추가
```
foreach (var neighborPos in _nodeManager.GetNeighbors(current.Pos))
{
    // 방문 기록이 있다면 다음 이웃 노드 확인
    if (_closedSet.Contains(neighborPos))
        continue;

    int sumG = current.G + _nodeManager.GetMoveCost(current.Pos);

    // _openDict에 이웃 노드 정보가 없으면 이웃 노드 생성 및 추가
    if (!_openDict.TryGetValue(neighborPos, out Node neighborNode))
    {
        neighborNode = _nodeManager.GetNode
        (
        neighborPos,
        sumG,
        Heuristic(neighborPos, targetCellPos),
        current
        );

        _openList.Add(neighborNode);
        _openDict.Add(neighborPos, neighborNode);

    }
    else if (sumG < neighborNode.G)
    {
        neighborNode.G = sumG;
        neighborNode.Parent = current;
    }
}
```
- 이웃노드가 _closedSet에 있다면 다음 이웃노드 확인(방문한 적이 있으면 추가 탐색 필요x)
- _openDict을 통해 탐색한 적 없는 좌표라면 노드를 새로 생성해서 오픈 리스트에 추가

##### 4) 탐색한 노드가 타겟 노드라면 해당 경로를 반환
```
private List<Vector2> BuildPath(Node targetNode)
{
    List<Vector2> path = new List<Vector2>();
    Node current = targetNode;

    while (current != null)
    {
        if(current.Parent != null)

        // 타일 중앙 좌표로 변환
        path.Add(GetCellPos(current.Pos));
        current = current.Parent;
    }

    path.Reverse();

    return path;
}
```
- 탐색 노드의 parent를 열람하면서 이전에 이동한 노드를 역추적한다.
- 역추적한 노드 리스트를 역정렬 시켜 처음 시작 노드 -> 목적 노드까지의 경로를 반환한다.

## A스타의 유의사항
#### 1. 무한 반복문이 생기지 않도록 방어코드를 설계해야 한다.
만약 사방이 막혀있는 물리적으로 갈 수 없는 노드를 타겟으로 지정할 경우, <br>
탐색 시행은 무한히 반복하게 되면서 유니티 내부적으로 무한반복문에 갇히게 된다. <br>
그렇기 때문에 반드시 방어코드를 설정해야 한다. <br>
(시행 횟수를 정하거나, 탐색 거리가 일정치 이상으로 멀어지면 멈추게 설계해야 한다.)

#### 2. 모든 변수들은 초기화를 해주어야 한다.
오픈리스트, 클로즈드셋등의 모든 참조타입은 캡슐화를 했기 때문에 시행이 끝날 때마다, <br>
반드시 초기화를 해주어서 다음 시행에 영향을 주어서는 절대 안 된다.

> ## 🤔 마치며
> A스타 실제 구현을 하며 이해하기 어려웠던 부분들이 많이 있었다. <br>
처음 A스타를 사용하기 위해 타일맵과 연동시키는 과정 이해하는 것이 먼저 필요했다. <br>
그 다음으로는 알고리즘의 동작 로직을 이해하는 것도 꽤 시간이 걸렸다. <br>
>
> 하지만 이 중에서 가장 문제가 되었던 것은 경로 탐색 오류와 무한 반복문이었다. <br>
먼저 경로 탐색 오류는 이미 셀좌표가 확인 된 노드가 Dictionary에 있을 때 g값에 따라 갱신을 해주는 방식인데, <br>
캡슐화를 하고 이를 각 시행때마다 **초기화하지 않았기 때문에** 이전 경로탐색 흔적이 버그를 일으켰다. <br>
이는 A스타 알고리즘뿐 아니라 모든 알고리즘을 설계할 때 주의해야 하는 사항이다.
>
> 그리고 무한 반복문은 **방어코드의 부재**로 인해 발생한 치명적인 문제였다. <br>
경로 탐색할 타겟 포지션이 고정 되어 있을 때는 갈 수 있는 위치로 설정하기 때문에 문제가 발생하지 않았다. <br>
하지만 런타임 중에 실시간으로 타겟을 새롭게 정해서 탐색을 하는 경우 이 문제가 발생했다. <br>
이 문제가 발생한 상황은 플레이어 추적 상태에서 플레이어의 위치를 기준으로 경로탐색을 계속해서 해나갈 때, <br>
셀좌표의 기준점과 실제 콜라이더상으로 갈 수 있는 타일의 경계가 다르기에 갈 수 없는 경로로 인지해버린 경우와 <br>
순찰상태에서 랜덤한 순찰 위치를 정할 때, 벽 타일로 둘러싸인 타일을 찾았을 때 무한 반복이 발생했다. <br>
이를 통해 방어코드가 얼마나 중요한지를 깨닫는 중요한 경험이었던 것 같다.
